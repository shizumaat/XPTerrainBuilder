"""Spec §62 (5) R-F, THE AIRSIDE-SEATED HALF — the 29ac fallback cap never
pairs an airside-seated pad's own vertex (owner RULINGS 2026-10-09d (1) /
08c (4); ``constraints/pavement_cap.py``).

The shape it stands for: a held terminal pad whose rim vertex the fallback
welded, two-sided at 10 % over 1.04 m, to a service-road vertex its own ramp
ceiling fixes — the LP relaxed the pad's hard plane instead (81 pad-tier
rows).  A LANDSIDE-ONLY pad keeps the weld: it is that pad's de-facto seat
until §62 (2)'s own seat lands.
"""
from __future__ import annotations

import sys
from pathlib import Path

import pytest

sys.path.insert(0, str(Path(__file__).resolve().parent))
from test_v2frontage import (HALF_W, RUN_LEN, _built, _cells, _rect,  # noqa: E402
                             _verts)

from auto_patch_v2.classify.roles import Cell  # noqa: E402
from auto_patch_v2.constraints import pavement_cap  # noqa: E402
from auto_patch_v2.constraints.pads import pad_fronts_airside  # noqa: E402
from auto_patch_v2.law import Law  # noqa: E402


@pytest.fixture(scope="module")
def law():
    return Law.for_airport("ZZZZ")


def _cross(pm, rows, pad_ref, other_ref):
    """The fallback rows between a vertex only ``pad_ref`` carries and a
    vertex of ``other_ref``."""
    pad_fids = {f.id for f in pm.faces.values() if f.ref.split("#")[0] == pad_ref}
    own = {v for v in _verts(pm, pad_ref)
           if set(pm.vertices[v].incident_faces) <= pad_fids}
    other = _verts(pm, other_ref)
    assert own and other
    return [r for r in rows if {r.a, r.b} & own and {r.a, r.b} & other]


def test_the_fallback_cap_never_pairs_an_airside_seated_pads_own_vertex(law):
    """The pad fronts the apron (§20 seats it); the road a metre off its
    north rim is §28's follower.  No fallback pair joins a vertex only the
    pad carries to the road — and the road's own ring stays capped."""
    pm, _airport = _built(law, _cells("service_road"))
    pad_fids = {f.id for f in pm.faces.values() if f.ref.split("#")[0] == "padA"}
    assert pad_fids & pad_fronts_airside(pm, law)
    rows = pavement_cap.pavement_road_cap([], pm, law)
    road = _verts(pm, "lotA")
    assert [r for r in rows if r.a in road and r.b in road], "the road's ring"
    assert _cross(pm, rows, "padA", "lotA") == []


def test_a_landside_only_pad_keeps_the_weld(law):
    """A pad that fronts nothing airside is not seated by §20: the fallback
    pair to the road a metre off its rim is unchanged."""
    cells = [Cell(0, "runway", "09/27",
                  _rect(-RUN_LEN / 2, -HALF_W, RUN_LEN / 2, HALF_W),
                  (), 3, "D", "airside", "runway", {}),
             Cell(1, "building", "padL", _rect(-60.0, 400.0, 60.0, 460.0), (),
                  None, None, "groundside", "pad", {}),
             Cell(2, "service_road", "roadL", _rect(-160.0, 389.0, 160.0, 399.5),
                  (), None, None, "groundside", "service_road", {})]
    pm, _airport = _built(law, cells)
    pad = next(f for f in pm.faces.values() if f.ref == "padL")
    assert pad.id not in pad_fronts_airside(pm, law)
    rows = pavement_cap.pavement_road_cap([], pm, law)
    assert _cross(pm, rows, "padL", "roadL"), "the de-facto seat stands"
